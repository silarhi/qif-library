import { Card, CardGrid } from '@astrojs/starlight/components'

```bash
composer require silarhi/qif-library
```

<CardGrid>
    <Card title="Parser" icon="document">
        Reads the date, amount, memo, cleared status, payee, address and category of every transaction.
    </Card>
    <Card title="Writer" icon="pencil">
        Builds QIF output from `Transaction` objects, including split transactions.
    </Card>
    <Card title="Typed results" icon="list-format">
        Dates as `DateTimeImmutable`, amounts as `float`, account types and cleared status as enums.
    </Card>
    <Card title="Lenient dates" icon="approve-check">
        Accepts `Y-m-d`, `d/m/Y`, `y-m-d` and `d/m/y`, the `'` year separator and space-padded dates (`1/ 1/ 0`).
    </Card>
    <Card title="No dependencies" icon="rocket">
        Plain PHP 8.2+. Works with Windows (`\r\n`) and Unix line endings, and strips the UTF-8 BOM.
    </Card>
</CardGrid>
